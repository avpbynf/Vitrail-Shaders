#version 460 core
#define <engine defines>

	mat4 of_ModelViewProjectionMatrix;
	vec4 of_DepthConv;
};
uniform sampler2D ofTexture;
in vec4 of_Vertex;
in vec4 of_Color;
in vec3 of_Normal;
in vec4 of_MultiTexCoord0;
invariant gl_Position;
out float of_FogFragCoord;
out vec4 owed;
flat out int owedFlat;


out vec2 texcoord;
out vec4 glcolor;
out vec3 kept;
 vec3 withheld;
out float fog;

void ofPackMain() {
	texcoord = of_MultiTexCoord0.xy;
	glcolor = of_Color;
	kept = of_Normal;
	withheld = of_Vertex.xyz;
	fog = of_FogFragCoord;
	gl_Position = of_ModelViewProjectionMatrix * of_Vertex;
}
void main() { owed = vec4(0); owedFlat = int(0); ofPackMain(); gl_Position.z = of_DepthConv.x * gl_Position.z + of_DepthConv.y * gl_Position.w; }

#define <engine defines>

	mat4 of_ModelViewProjectionMatrix;
	vec4 of_DepthConv;
};

	ofFragData0 = c + float(owedFlat);
}
