	mat4 of_ModelViewMatrix;
	mat4 of_ProjectionMatrix;

	mat4 gbufferProjectionInverse;
	vec4 of_DepthConv;

in vec4 of_Vertex;
in vec4 of_Color;
in vec4 of_MultiTexCoord0;
invariant gl_Position;




in vec4 mc_Entity;
out vec2 texcoord;
out vec4 glcolor;

void ofPackMain() {
	texcoord = of_MultiTexCoord0.xy;
	glcolor = of_Color;
	vec4 pos = of_ModelViewMatrix * of_Vertex;
	gl_Position = of_ProjectionMatrix * pos;
	gl_Position.z *= 0.999;
	if (mc_Entity.x == 10.0) gl_Position.xy += 0.001;
}
void main() { ofPackMain(); gl_Position.z = of_DepthConv.x * gl_Position.z + of_DepthConv.y * gl_Position.w; }
