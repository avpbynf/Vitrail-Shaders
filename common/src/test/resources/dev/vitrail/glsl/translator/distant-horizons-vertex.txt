#define of_MultiTexCoord0 vec4(0.0, 0.0, 0.0, 1.0)
#define of_MultiTexCoord1 vec4(ofDistantLight(), 0.0, 1.0)
#define of_MultiTexCoord2 vec4(ofDistantLight(), 0.0, 1.0)

#define of_MultiTexCoord6 vec4(0.0, 0.0, 0.0, 1.0)

vec2 ofDistantLight() {
	return (vec2(float((meta >> 4u) & 15u), float(meta & 15u)) + 0.5) / 16.0;
}
vec3 ofDistantNudge() {
	uint at = (meta >> 8u) & 63u;
	float x = (at & 1u) != 0u ? 0.01 : 0.0;
	float z = (at & 16u) != 0u ? 0.01 : 0.0;
	return vec3((at & 2u) != 0u ? -x : x, 0.0, (at & 32u) != 0u ? -z : z);
}
vec4 mc_Entity = vec4(0.0);
void ofDistantVertex() {
	of_Vertex = vec4(vec3(vPosition) + of_SectionOffset + ofDistantNudge(), 1.0);
	of_Color = vColor;
	of_Normal = ofDistantFaces[irisNormal];
}
invariant gl_Position;



out vec2 texcoord;
out vec2 lmcoord;
out vec4 glcolor;

void ofPackMain() {
	texcoord = (mat4(1.0) * of_MultiTexCoord0).xy;
	lmcoord = (mat4(1.0) * of_MultiTexCoord2).xy + (mat4(1.0) * of_MultiTexCoord1).xy;
	glcolor = of_Color;
	gl_Position = of_ModelViewProjectionMatrix * of_Vertex;
}
void main() { ofDistantVertex(); ofPackMain(); gl_Position.z = of_DepthConv.x * gl_Position.z + of_DepthConv.y * gl_Position.w; }
