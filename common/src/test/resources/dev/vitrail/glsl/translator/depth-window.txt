#version 460 core
#define <engine defines>
layout(std140) uniform OfGlobals {
	vec4 of_DepthConv;
	float near;
	float far;
};
uniform sampler2D depthtex0;
uniform sampler2D colortex2;
layout(location = 0) out vec4 ofFragData0;
void ofOrderOutputs() { ofFragData0; }






in vec2 texcoord;

float linearDepth(float d) {
	return (2.0 * near) / (far + near - d * (far - near));
}

void main() { ofOrderOutputs();
	float z = (of_DepthConv.z * gl_FragCoord.z + of_DepthConv.w);
	vec3 screen = vec3(gl_FragCoord.xy, of_DepthConv.z * gl_FragCoord.z + of_DepthConv.w);
	vec2 pixel = gl_FragCoord.xy;
	vec2 tail = gl_FragCoord.zw;
	float w = gl_FragCoord.w;
	vec4 whole = gl_FragCoord;
	float d = textureLod(depthtex0, texcoord, 0.0).r + textureLod(colortex2, texcoord, 0.0).r;
	gl_FragDepth = (of_DepthConv.z * ( linearDepth(z) + d) + of_DepthConv.w);
	gl_FragDepth += 0.001;
	ofFragData0 = vec4(screen, pixel.x + w + whole.x + tail.x);
}
