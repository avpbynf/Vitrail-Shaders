layout(location = 0) out vec4 ofFragData0;
void ofOrderOutputs() { ofFragData0; }







in vec2 uv;

void main() { ofOrderOutputs();
	vec4 a = textureLod(colortex0, uv, 0.0);
	vec4 b = textureLod(colortex0, uv,0.0);
	vec4 c = textureProj(colortex0, vec3(uv, 1.0));
	vec4 d = textureLod(colortex0, uv, 0.0);
	vec4 e = textureLod(colortex0, uv, 0.0);
	vec4 f = textureLodOffset(colortex0, uv, 0.0, ivec2(1, 0));
	vec4 g = textureLodOffset(colortex0, uv,0.0, ivec2(0, 1));
	vec4 h = texelFetch(colortex0, ivec2(gl_FragCoord.xy), 0);
	vec4 i = textureLod(sky, vec3(uv, 1.0), 0.0);
	vec4 j = textureLod(ramp, uv.x, 0.0) + textureLod(lut, vec3(uv, 0.5), 0.0) + textureLod(lut, vec3(uv, 0.5),0.0);
	vec4 k = textureProjLod(colortex0, vec3(uv, 1.0), 0.0) + texelFetch(lut, ivec3(1), 0);
	float depth = textureLod(depthtex0, uv, 0.0).r;
	ofFragData0 = a + b + c + d + e + f + g + h + i + j + k + depth;
}
