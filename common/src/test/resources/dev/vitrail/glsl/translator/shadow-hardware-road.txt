float shadowFilter(vec3 pos) {
	float sum = 0.0;
	sum += vec4(textureLod(shadowtex0, pos, 0.0)).x;
	sum += vec4(textureLod(shadowtex1, pos + vec3(0.001, 0.0, 0.0), 0.0)).r;
	sum += vec4(textureLod(shadowtex0, pos,0.0)).x;
	sum += vec4(textureProj(shadowtex0, vec4(pos, 1.0))).x;
	sum += textureLod(shadowtex1HW, pos, 0.0);
	sum += textureLod(shadowtex0, pos,0.0);
	sum += textureLodOffset(shadowtex1, pos, 0.0, ivec2(1, 0));
	sum += textureProj(shadowtex0, vec4(pos, 1.0));
	return sum * 0.125;
}

void main() { ofOrderOutputs();
	ofFragData0 = textureLod(colortex0, texcoord, 0.0) * shadowFilter(shadowPos);
}
