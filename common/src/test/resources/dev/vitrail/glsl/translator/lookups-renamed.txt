#version 460 core
#define <engine defines>
uniform sampler2D colortex0;
uniform sampler2D depthtex0;
uniform sampler3D lut;
uniform samplerCube sky;
uniform sampler1D ramp;
layout(location = 0) out vec4 ofFragData0;
void ofOrderOutputs() { ofFragData0; }







in vec2 uv;

void main() { ofOrderOutputs();
	vec4 a = texture(colortex0, uv);
	vec4 b = textureLod(colortex0, uv, 2.0);
	vec4 c = textureProj(colortex0, vec3(uv, 1.0));
	vec4 d = textureGrad(colortex0, uv, dFdx(uv), dFdy(uv));
	vec4 e = textureGrad(colortex0, uv, vec2(0.0), vec2(0.0));
	vec4 f = textureOffset(colortex0, uv, ivec2(1, 0));
	vec4 g = textureLodOffset(colortex0, uv, 1.0, ivec2(0, 1));
	vec4 h = texelFetch(colortex0, ivec2(gl_FragCoord.xy), 0);
	vec4 i = texture(sky, vec3(uv, 1.0));
	vec4 j = texture(ramp, uv.x) + texture(lut, vec3(uv, 0.5)) + textureLod(lut, vec3(uv, 0.5), 1.0);
	vec4 k = textureProjLod(colortex0, vec3(uv, 1.0), 0.0) + texelFetch(lut, ivec3(1), 0);
	float depth = texture(depthtex0, uv).r;
	ofFragData0 = a + b + c + d + e + f + g + h + i + j + k + depth;
}
