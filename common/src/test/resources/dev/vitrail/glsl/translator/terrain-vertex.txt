float ofAxis(uint at) {
	uint top = (a_Position.x >> at) & 1023u;
	uint bottom = (a_Position.y >> at) & 1023u;
	return float(top * 1024u + bottom) * (32.0 / 1048576.0) - 8.0;
}
vec3 ofSectionOrigin(uint index) {
	return vec3(float((index >> 5u) & 7u), float(index & 3u), float((index >> 2u) & 7u)) * 16.0;
}
vec3 ofFrameNormal(uint word) {
	vec2 folded = vec2(bitfieldExtract(int(word), 0, 12), bitfieldExtract(int(word), 12, 11)) * vec2(1.0 / 2047.0, 1.0 / 1023.0);
	float up = 1.0 - abs(folded.x) - abs(folded.y);
	vec2 sides = vec2(folded.x >= 0.0 ? 1.0 : -1.0, folded.y >= 0.0 ? 1.0 : -1.0);
	return normalize(vec3(up >= 0.0 ? folded : (1.0 - abs(folded.yx)) * sides, up));
}
vec4 ofFrameTangent(vec3 normal, uint word) {
	float side = normal.z >= 0.0 ? 1.0 : -1.0;
	float scale = -1.0 / (side + normal.z);
	float across = normal.x * normal.y * scale;
	vec3 first = vec3(1.0 + side * normal.x * normal.x * scale, side * across, -side * normal.x);
	vec3 second = vec3(across, side + normal.y * normal.y * scale, -normal.y);
	float turn = float(word >> 24u) * (1.0 / 256.0);
	float along = 1.0 - 4.0 * abs(turn - 0.5);
	float away = (turn >= 0.5 ? 1.0 : -1.0) * (1.0 - abs(along));
	return vec4(normalize(along * first + away * second), ((word >> 23u) & 1u) != 0u ? 1.0 : -1.0);
}
void ofSodiumVertex() {
	vec3 ofLocal = vec3(ofAxis(0u), ofAxis(10u), ofAxis(20u));
	of_Vertex = vec4(ofLocal + of_RegionOffset + ofSectionOrigin(a_LightAndData.w), 1.0);
	of_Color = a_Color;
	vec2 ofInward = vec2(a_TexCoord >> 15u) * 2.0 - 1.0;
	of_MultiTexCoord0 = vec4(vec2(a_TexCoord & 32767u) / 32768.0 + ofInward * of_TexShrink, 0.0, 1.0);
	of_MultiTexCoord1 = vec4(vec2(a_LightAndData.xy), 0.0, 1.0);
	of_Normal = ofFrameNormal(a_TangentFrame);
	mc_Entity = vec4(int(a_BlockId >> 1u) - 1, a_BlockId & 1u, 0.0, 1.0);
	mc_midTexCoord = vec4(vec2(a_MidTexCoord) * (1.0 / 32768.0), 0.0, 1.0);
	at_tangent = ofFrameTangent(of_Normal, a_TangentFrame);
	at_midBlock = vec3(a_MidBlock.xyz);
}

void ofPackMain() {
	vec4 position = of_ModelViewMatrix * of_Vertex;
	vec3 world = (gbufferModelViewInverse * position).xyz + cameraPosition;
	texcoord = (of_TextureMatrix[0] * of_MultiTexCoord0).xy;
	lmcoord = (of_TextureMatrix[1] * of_MultiTexCoord1).xy;
	glcolor = of_Color;
	normal = of_NormalMatrix * of_Normal;
	blockId = mc_Entity.x + at_midBlock.y + mc_midTexCoord.x + at_tangent.w + chunkOffset.x;
	gl_Position = of_ProjectionMatrix * position;
}
void main() { ofSodiumVertex(); ofPackMain(); gl_Position.z = of_DepthConv.x * gl_Position.z + of_DepthConv.y * gl_Position.w; }
