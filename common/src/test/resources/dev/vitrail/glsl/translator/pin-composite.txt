layout(location = 0) out vec4 ofFragData0;
void ofOrderOutputs() { ofFragData0; }











in vec2 texcoord;

 bool colortex1MipmapEnabled = true;
 bool colortex4MipmapEnabled = false;

vec3 bloom(vec2 uv) {
	vec3 sum = vec3(0.0);
	for (int i = -2; i <= 2; i++) {
		sum += texture(colortex1, uv + vec2(float(i) / viewWidth, 0.0), 2.0).rgb;
		sum += textureLod(colortex1, uv, float(i)).rgb;
	}
	return sum / 5.0;
}

void main() { ofOrderOutputs();
	vec3 colour = textureLod(colortex0, texcoord, 0.0).rgb;
	colour += textureLod(colortex0, texcoord, 0.0).rgb;
	colour += textureLodOffset(colortex0, texcoord, 0.0, ivec2(1, 1)).rgb;
	colour += textureLodOffset(colortex0, texcoord,0.0, ivec2(2, 0)).rgb;
	colour += textureLodOffset(colortex0, texcoord,0.0, ivec2(0, 1)).rgb;
	colour += textureLod(colortex4, texcoord, 0.0).rgb;
	colour += textureLod(depthtex1, texcoord, 0.0).rgb + textureLod(noisetex, texcoord * 8.0, 0.0).rgb;
	colour += texture(frameRect, gl_FragCoord.xy).rgb + textureLod(customMask, texcoord, 0.0).rgb;
	colour += bloom(texcoord) + textureProj(colortex0, vec3(texcoord, 1.0)).rgb;
	colour += textureGather(colortex0, texcoord, 1).rgb + texelFetch(colortex0, ivec2(3), 0).rgb;
	ofFragData0 = vec4(colour, 1.0);
}
