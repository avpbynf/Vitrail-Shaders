#version 460 core
#define <engine defines>
layout(std140) uniform OfGlobals {
	float frameTimeCounter;
	int frameCounter;
	int isEyeInWater;
	int entityId;
	vec3 cameraPosition;
	vec2 resolution[2];
	mat4 gbufferModelView;
	mat4 gbufferProjection;
	ivec2 atlasSize;
	float conflict;
	vec4 entityColor;
	float multiline;
	float second;
};
uniform sampler2D colortex0;
uniform sampler2D colortex1;
layout(location = 0) out vec4 ofFragData0;
void ofOrderOutputs() { ofFragData0; }











#ifdef HAS_FOG
uniform float fogDensity; // @dead
#else
float fogDensity = 0.5;
#endif




in vec2 texcoord;

void main() { ofOrderOutputs();
	vec4 c = textureLod(colortex0, texcoord, 0.0) + textureLod(colortex1, texcoord, 0.0);
	float f = frameTimeCounter + float(frameCounter + isEyeInWater + entityId) + multiline + second;
	ofFragData0 = c * f * fogDensity + entityColor + vec4(cameraPosition, float(atlasSize.x));
}
