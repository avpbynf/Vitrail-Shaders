--- shadow-compute.csh (COMPUTE) ---
#version 460 core
#define <engine defines>
layout(std140) uniform OfGlobals {
	mat4 shadowProjection;
};
uniform sampler2D shadowtex0;
layout(rgba8) writeonly uniform image2D colorimg0;
layout(r32f) coherent uniform image3D voxelVolume;
float ofShadowCompare(sampler2D ofMap, vec3 ofAt, ivec2 ofOffset) { vec2 ofSize = vec2(textureSize(ofMap, 0)); vec2 ofUv = ofAt.xy + vec2(ofOffset) / ofSize; vec4 ofTests = step(vec4(ofAt.z), textureGather(ofMap, ofUv, 0)); vec2 ofPart = fract(ofUv * ofSize - 0.5); return mix(mix(ofTests.w, ofTests.z, ofPart.x), mix(ofTests.x, ofTests.y, ofPart.x), ofPart.y); }
float ofShadowCompare(sampler2D ofMap, vec3 ofAt) { return ofShadowCompare(ofMap, ofAt, ivec2(0)); }
float ofShadowCompare(sampler2D ofMap, vec3 ofAt, float ofLod) { return ofShadowCompare(ofMap, ofAt, ivec2(0)); }
float ofShadowCompare(sampler2D ofMap, vec3 ofAt, float ofLod, ivec2 ofOffset) { return ofShadowCompare(ofMap, ofAt, ofOffset); }
float ofShadowCompare(sampler2D ofMap, vec3 ofAt, ivec2 ofOffset, float ofBias) { return ofShadowCompare(ofMap, ofAt, ofOffset); }


layout(local_size_x = 8, local_size_y = 8) in;






void main() {
	ivec2 texel = ivec2(gl_GlobalInvocationID.xy);
	float lit = ofShadowCompare(shadowtex0, vec3(vec2(texel) / 64.0, 0.5), 0.0, ivec2(1, 0));
	imageStore(colorimg0, texel, vec4(lit));
	imageStore(voxelVolume, ivec3(texel, 0), vec4(lit));
}
