vec4 ofTexture3D_colortex6(sampler2D ofMap, vec3 ofAt) {
	vec3 ofQ = fract(ofAt);
	float ofZ = ofQ.z * 32.0 - 0.5;
	float ofBase = floor(ofZ);
	vec2 ofIn = ofQ.xy * vec2(32.0, 32.0) + 1.0;
	int ofNear = int(mod(ofBase, 32.0));
	int ofFar = int(mod(ofBase + 1.0, 32.0));
	vec2 ofTile = vec2(34.0, 34.0);
	vec2 ofSize = vec2(204.0, 204.0);
	vec2 ofA = (vec2(ofNear % 6, ofNear / 6) * ofTile + ofIn) / ofSize;
	vec2 ofB = (vec2(ofFar % 6, ofFar / 6) * ofTile + ofIn) / ofSize;
	return mix(texture(ofMap, ofA), texture(ofMap, ofB), clamp(ofZ - ofBase, 0.0, 1.0));
}
vec4 ofTexture3D_colortex6(sampler2D ofMap, vec3 ofAt, float ofLevel) {
	return ofTexture3D_colortex6(ofMap, ofAt);
}

	ofFragData0 = vec4(a, b, c, textureLod(colortex0, texcoord, 0.0).a);
}
