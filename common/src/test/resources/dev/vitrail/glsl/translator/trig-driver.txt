#version 460 core
#define <engine defines>
layout(std140) uniform OfGlobals {
	mat4 of_ModelViewProjectionMatrix;
	float frameTimeCounter;
	vec4 of_DepthConv;
};
in vec4 of_Vertex;
invariant gl_Position;



out vec3 wave;

vec2 rotate(vec2 p, float a) {
	return vec2(cos(a) * p.x - sin(a) * p.y, sin(a) * p.x + cos(a) * p.y);
}

void ofPackMain() {
	vec3 p = of_Vertex.xyz;
	float t = frameTimeCounter * 0.5;
	wave = vec3(sin(p.x + t), cos(p.z * 2.0 - t), sin(length(p.xz)) * cos(t));
	p.xz = rotate(p.xz, t);
	gl_Position = of_ModelViewProjectionMatrix * vec4(p, 1.0);
}
void main() { ofPackMain(); gl_Position.z = of_DepthConv.x * gl_Position.z + of_DepthConv.y * gl_Position.w; }
