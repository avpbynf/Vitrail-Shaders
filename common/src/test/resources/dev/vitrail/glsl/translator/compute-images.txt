#version 460 core
#define <engine defines>
layout(std140) uniform OfGlobals {
	vec2 viewSize;
};
layout(rgba16f) uniform image2D colorimg1;
layout(r32ui) readonly uniform uimage3D voxels;
layout(rgba8) writeonly uniform image2D colorimg0;
coherent restrict volatile uniform iimage2D counters;
uniform sampler2D colortex0;


layout(local_size_x = 16, local_size_y = 16, local_size_z = 1) in;
const ivec3 workGroups = ivec3(32, 32, 1);








void main() {
	ivec2 texel = ivec2(gl_GlobalInvocationID.xy);
	vec4 c = imageLoad(colorimg1, texel);
	imageStore(colorimg0, texel, c * texture(colortex0, vec2(texel) / viewSize));
}
