out vec4 glcolor;
out vec3 normal;

void ofPackMain() {
	vec4 position = of_ModelViewMatrix * of_Vertex;
	gl_Position = of_ProjectionMatrix * position;
	texcoord = (of_TextureMatrix[0] * of_MultiTexCoord0).xy;
	lmcoord = (of_TextureMatrix[1] * of_MultiTexCoord1).xy;
	glcolor = of_Color;
	normal = normalize(of_NormalMatrix * of_Normal);
	of_FogFragCoord = length(position.xyz);
	of_FrontColor = of_Color;
	vec4 clip = of_ModelViewProjectionMatrix * of_Vertex;
	if (clip.w < 0.0 && gl_VertexIndex == 0) gl_Position = clip;
}
void main() { ofPackMain(); gl_Position.z = of_DepthConv.x * gl_Position.z + of_DepthConv.y * gl_Position.w; }
