#version 460 core
#define <engine defines>
struct OfFog { vec4 color; float density; float start; float end; float scale; };
layout(std140) uniform OfGlobals {
	OfFog of_Fog;
	vec3 fogColor;
};
uniform sampler2D ofTexture;
uniform sampler2D lightmap;
in float of_FogFragCoord;
layout(location = 0) out vec4 ofFragData0;
layout(location = 1) out vec4 ofFragData1;
layout(location = 2) out vec4 ofFragData2;
void ofOrderOutputs() { ofFragData0; ofFragData1; ofFragData2; }






in vec2 texcoord;
in vec2 lmcoord;
in vec4 glcolor;
in vec3 normal;

/* DRAWBUFFERS:012 */

void main() { ofOrderOutputs();
	vec4 albedo = texture(ofTexture, texcoord) * glcolor;
	albedo.rgb *= texture(lightmap, lmcoord).rgb;
	albedo.rgb = mix(albedo.rgb, fogColor, clamp(of_FogFragCoord * of_Fog.density, 0.0, 1.0));
	ofFragData0 = albedo;
	ofFragData1 = vec4(normal * 0.5 + 0.5, 1.0);
	ofFragData2 = vec4(lmcoord, 0.0, 1.0);
}
