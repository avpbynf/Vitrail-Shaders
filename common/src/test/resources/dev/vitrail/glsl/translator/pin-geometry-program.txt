#version 460 core
#define <engine defines>
uniform sampler2D ofTexture;
uniform sampler2D lightmap;
uniform sampler2D normals;
uniform sampler2D specular;
uniform sampler2D colortex4;
uniform sampler2D noisetex;
uniform sampler2D depthtex1;
uniform sampler2D shadowcolor0;
layout(location = 0) out vec4 ofFragData0;
void ofOrderOutputs() { ofFragData0; }










in vec2 texcoord;

void main() { ofOrderOutputs();
	vec4 albedo = texture(ofTexture, texcoord);
	vec3 n = texture(normals, texcoord).rgb + texture(specular, texcoord).rgb;
	vec3 l = texture(lightmap, texcoord).rgb;
	vec3 g = textureLod(colortex4, texcoord, 0.0).rgb + textureLod(noisetex, texcoord, 0.0).rgb;
	g += textureLod(depthtex1, texcoord, 0.0).rgb + textureLod(shadowcolor0, texcoord, 0.0).rgb;
	ofFragData0 = albedo * vec4(n + l + g, 1.0);
}
